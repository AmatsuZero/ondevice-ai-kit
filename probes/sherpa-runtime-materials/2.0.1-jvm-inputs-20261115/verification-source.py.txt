def sha(data): return hashlib.sha256(data).hexdigest()
def blob(data): return hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()
def load(p): return json.loads(p.read_bytes())
def save(p, obj): p.write_text(json.dumps(obj, indent=2, ensure_ascii=False) + '\n')
def identity(p):
    data = p.read_bytes()
    return {'path': str(p.relative_to(ROOT)), 'bytes': len(data), 'sha256': sha(data)}
def run(name, args):
    p = subprocess.run(['rtk', 'proxy', *args], cwd=ROOT, capture_output=True, timeout=55)
    (OPS / (name + '.stdout')).write_bytes(p.stdout)
    (OPS / (name + '.stderr')).write_bytes(p.stderr)
    assert p.returncode == 0, (name, p.returncode)
    return p.stdout

treepath = ROOT / '.omx/artifacts/w3-model-reselection-20261110/sherpa-v201-tree.raw'
tree = load(treepath)
resolution = load(OUT / 'source-link-resolution.json')
assert sha(treepath.read_bytes()) == resolution['treeSHA256']
assert tree['truncated'] is False and tree['sha'] == resolution['sourceCommit']
entries = {r['path']: r for r in tree['tree']}
links = resolution['links']
assert len(links) == 22
for r in links:
    link = (ROOT / r['linkEvidencePath']).read_bytes()
    source = (ROOT / r['resolvedEvidencePath']).read_bytes()
    original = entries[r['sourcePath']]
    target = entries[r['resolvedPath']]
    assert original['mode'] == r['sourceMode'] == '120000'
    assert target['mode'] == r['resolvedMode'] == '100644'
    assert original['sha'] == r['gitBlobSHA1'] == blob(link)
    assert target['sha'] == r['resolvedGitBlobSHA1'] == blob(source)
    assert sha(link) == r['linkSHA256'] and sha(source) == r['resolvedSHA256']
    assert len(link) == r['linkBytes'] == original['size']
    assert len(source) == r['resolvedBytes'] == target['size']
    assert link.decode() == r['linkText']
    assert posixpath.normpath(posixpath.join(posixpath.dirname(r['sourcePath']), link.decode())) == r['resolvedPath']
    for rec, expected in ((r, link), (r['sourceAcquisition'], source)):
        if 'apiResponsePath' in rec:
            raw = (ROOT / rec['apiResponsePath']).read_bytes()
            assert sha(raw) == rec['apiResponseSHA256']
            response = json.loads(raw)
            assert base64.b64decode(response['content']) == expected
        else:
            assert (ROOT / rec['reusedExistingPath']).read_bytes() == expected
