from pathlib import Path, PurePosixPath
import difflib, hashlib, json, shutil, subprocess, tarfile

ROOT = Path('/Users/daubert/Github/ondevice-ai-kit/.worktrees/foundation')
OPS = ROOT / '.omx/artifacts/w3-sentencepiece-patch-20261115'
OUT = ROOT / 'probes/sherpa-runtime-materials/2.0.1-sentencepiece-patch-20261115'
PLAN = json.loads((OPS / 'plan.json').read_text())
def sha(data): return hashlib.sha256(data).hexdigest()
def blob(data): return hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()
def save(path, value): path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')
def identity(path):
    raw = path.read_bytes()
    return {'path': str(path.relative_to(ROOT)), 'bytes': len(raw), 'sha256': sha(raw)}
def run(name, args):
    p = subprocess.run(['rtk', 'proxy', *args], cwd=OPS, capture_output=True, timeout=30)
    (OPS / (name + '.stdout')).write_bytes(p.stdout)
    (OPS / (name + '.stderr')).write_bytes(p.stderr)
    save(OPS / (name + '-process.json'), {'exitCode': p.returncode, 'args': args})
    assert p.returncode == 0, (name, p.returncode)
    return p.stdout

module = ROOT / PLAN['sourceModulePath']
module_data = module.read_bytes()
assert sha(module_data) == PLAN['sourceModuleSHA256']
fixed_tree = json.loads((ROOT / '.omx/artifacts/w3-model-reselection-20261110/sherpa-v201-tree.raw').read_bytes())
assert fixed_tree['sha'] == PLAN['sourceCommit'] and fixed_tree['truncated'] is False
module_entry = next(r for r in fixed_tree['tree'] if r['path'] == 'cmake/simple-sentencepiece.cmake')
assert module_entry['mode'] == '100644' and module_entry['sha'] == blob(module_data)
component_tree = json.loads((ROOT / PLAN['originalTreePath']).read_bytes())
assert component_tree['sha'] == '62dd423df1d51da5ea06f1c3a046fc04f01b4f39'
assert component_tree['truncated'] is False
entries = {r['path']: r for r in component_tree['tree']}
archive = OPS / 'simple-sentencepiece-v0.7.tar.gz'
assert sha(archive.read_bytes()) == PLAN['expectedArchiveSHA256']
original = {}
archive_files = []
with tarfile.open(archive, 'r:gz') as tf:
    members = tf.getmembers()
    names = [m.name for m in members]
    assert len(names) == len(set(names))
    for member in members:
        path = PurePosixPath(member.name)
        assert not path.is_absolute() and '..' not in path.parts
        relative = '/'.join(path.parts[1:])
        if member.isfile():
            data = tf.extractfile(member).read()
            assert relative in entries and entries[relative]['type'] == 'blob'
            assert entries[relative]['mode'] in ('100644', '100755')
            assert blob(data) == entries[relative]['sha'] and len(data) == entries[relative]['size']
            archive_files.append(relative)
            if relative in PLAN['files'] or relative == 'LICENSE':
                original[relative] = data
    expected_regular = {r['path'] for r in component_tree['tree']
                        if r['type'] == 'blob' and r['mode'] in ('100644', '100755')}
    assert set(archive_files) == expected_regular
assert set(original) == set(PLAN['files']) | {'LICENSE'}
for name in PLAN['files']:
    before = OUT / 'original' / name
    patched = OUT / 'patched' / name
    before.parent.mkdir(parents=True, exist_ok=True)
    patched.parent.mkdir(parents=True, exist_ok=True)
    before.write_bytes(original[name])
    patched.write_bytes(original[name])

text = module_data.decode()
helper_start = text.index('function(write_file_if_different path content)')
helper_end = text.index('endfunction()', helper_start) + len('endfunction()')
body_start = text.index('    # On Termux,', text.index('  elseif(CMAKE_SYSTEM_NAME STREQUAL Android)'))
message = '    message(STATUS "Patched ssentencepiece for Android: all threading removed")'
body_end = text.index(message, body_start) + len(message)
helper = text[helper_start:helper_end]
body = text[body_start:body_end]
script = ('# Exact selected helper and Android patch body; no FetchContent/configure/build.\n'
          'set(simple-sentencepiece_SOURCE_DIR "' + str(OUT / 'patched') + '")\n' + helper + '\n' + body + '\n')
script_path = OPS / 'apply-android-patch.cmake'
script_path.write_text(script)
cmake = Path('/Users/daubert/Library/Android/sdk/cmake/3.22.1/bin/cmake')
assert cmake.is_file()
version = run('cmake-version', [str(cmake), '--version']).decode().splitlines()[0]
assert '3.22.1' in version
stdout = run('apply-android-patch', [str(cmake), '-P', str(script_path)])
assert stdout.count(b'Patched ssentencepiece for Android: all threading removed') == 1

rows = []
diff = ''
for name in PLAN['files']:
    before = OUT / 'original' / name
    after = OUT / 'patched' / name
    assert before.read_bytes() != after.read_bytes(), name
    diff += ''.join(difflib.unified_diff(before.read_text().splitlines(keepends=True),
                                        after.read_text().splitlines(keepends=True),
                                        fromfile='original/' + name, tofile='patched/' + name))
    rows.append({'path': name, 'sourceTreeGitBlobSHA1': entries[name]['sha'],
                 'before': {**identity(before), 'gitBlobSHA1': blob(before.read_bytes())},
                 'reconstructedAfter': {**identity(after), 'gitBlobSHA1': blob(after.read_bytes())}})
assert len(rows) == 3
header = (OUT / 'patched/ssentencepiece/csrc/ssentencepiece.h').read_text()
source = (OUT / 'patched/ssentencepiece/csrc/ssentencepiece.cc').read_text()
assert 'ThreadPool' not in header and 'pool_' not in header and 'std::thread' not in header
assert 'pool_->enqueue' not in source and 'std::future<' not in source and 'result.get()' not in source
(OUT / 'android-threading-removal.diff').write_text(diff)
(OUT / 'LICENSE').write_bytes(original['LICENSE'])
old_license = ROOT / 'probes/sherpa-runtime-materials/2.0.1-a8d09d1d/licenses/simple-sentencepiece/LICENSE'
assert old_license.read_bytes() == original['LICENSE']

groups_path = ROOT / 'probes/sherpa-runtime-materials/2.0.1-member-inputs-20261115/build-input-groups.json'

matrix_path = ROOT / 'probes/sherpa-runtime-materials/2.0.1-member-inputs-20261115/member-matrix.json'

lines = log_path.read_text().splitlines()

abi_rows = []
for index, group in enumerate(native_groups):
    start = group['actualCmake']['line'] - 1
    end = native_groups[index + 1]['actualCmake']['line'] - 1 if index + 1 < len(native_groups) else len(lines)
    needles = ('Downloading simple-sentencepiece https://github.com/pkufool/simple-sentencepiece/archive/refs/tags/v0.7.tar.gz',
               'Patched ssentencepiece for Android: all threading removed',
               'ssentencepiece_core.dir/ssentencepiece.cc.o',
               'Linking CXX static library ../../../../lib/libssentencepiece_core.a',
               'Built target ssentencepiece_core')
    observed = [{'line': i + 1, 'text': lines[i]} for i in range(start, end)
                if any(needle in lines[i] for needle in needles)]
    assert all(any(needle in r['text'] for r in observed) for needle in needles), group['id']
    assert sum('Patched ssentencepiece for Android: all threading removed' in r['text'] for r in observed) == 1
    abi = group['actualCmake']['abi']
    members = [r for r in matrix['rows'] if r['path'].startswith('jni/' + abi + '/')
               and not r['path'].endswith('/libonnxruntime.so')]
    assert len(members) == 3
    abi_rows.append({'abi': abi, 'inputGroup': group['id'], 'logEvidence': observed,
                     'correspondingSherpaMembers': [{k: r[k] for k in ('path', 'bytes', 'sha256')} for r in members],
                     'declaredLinkChain': ['ssentencepiece_core -> sherpa-onnx-core',
                         'sherpa-onnx-core -> sherpa-onnx-jni / sherpa-onnx-c-api',
                         'sherpa-onnx-c-api -> sherpa-onnx-cxx-api'],
                     'exactRetainedMachineCodeVerified': False})
cmake_paths = ['sherpa-onnx/csrc/CMakeLists.txt', 'sherpa-onnx/jni/CMakeLists.txt', 'sherpa-onnx/c-api/CMakeLists.txt']
link_evidence = []
for path in cmake_paths:

    assert blob(p.read_bytes()) == entry['sha']
    snippets = [{'line': i + 1, 'text': line} for i, line in enumerate(p.read_text().splitlines())
                if ('target_link_libraries(sherpa-onnx-core' in line or 'ssentencepiece_core' in line

                    or 'target_link_libraries(sherpa-onnx-c-api sherpa-onnx-core)' in line
                    or 'target_link_libraries(sherpa-onnx-cxx-api sherpa-onnx-c-api)' in line)]
    link_evidence.append({'source': identity(p), 'gitBlobSHA1': entry['sha'], 'lines': snippets})

with tarfile.open(archive, 'r:gz') as tf:
    for component, source_name, license_name, role in (
