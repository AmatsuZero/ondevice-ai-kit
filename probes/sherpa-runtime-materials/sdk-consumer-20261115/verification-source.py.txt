ROOT = Path(__file__).resolve().parents[3]
OUT = Path(__file__).resolve().parent
DELIVERY = ROOT / 'artifacts/verification/w3-sdk-consumer-binding-20261115'
M2 = DELIVERY / 'm2'
MATERIAL = ROOT / 'probes/sherpa-runtime-materials/sdk-consumer-20261115'

MODULES = ('sdk-core', 'speech', 'backend-sherpa-asr', 'backend-system-asr', 'backend-system-tts')


def sha(data, name='sha256'):
    return hashlib.new(name, data).hexdigest()


def identity(path):
    path = Path(path)
    data = path.read_bytes()
    try:
        name = str(path.relative_to(ROOT))
    except ValueError:
        name = str(path).replace(str(Path.home() / '.gradle'), '$GRADLE_USER_HOME')
    return {'path': name, 'bytes': len(data), 'sha256': sha(data)}


def emit(name, value):
    encoded = (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()
    (OUT / name).write_bytes(encoded)
    (MATERIAL / name).write_bytes(encoded)


assert subprocess.check_output(['rtk', 'proxy', 'git', 'rev-parse', 'HEAD'], cwd=ROOT).decode().strip() == SOURCE_HEAD
MATERIAL.mkdir(exist_ok=True)
sources = json.loads((OUT / 'source-inputs-before.json').read_text())
for name, expected in sources.items():
    actual = identity(ROOT / name)
    assert actual['bytes'] == expected['bytes'] and actual['sha256'] == expected['sha256'], name

processes = {}
task_counts = {}
for prefix, stream in [('sdk-publish', 'sdk-publish'), ('consumer', 'consumer-build'),
                       ('consumer-debug', 'consumer-debug-build')]:
    process = json.loads((OUT / (prefix + '-process.json')).read_text())
    assert process['exitCode'] == 0
    log = (OUT / (stream + '.stdout')).read_text()
    assert 'BUILD SUCCESSFUL' in log
    counts = re.search(r'(\d+) actionable tasks: (\d+) executed, (\d+) up-to-date', log)
    assert counts
    task_counts[prefix] = dict(zip(('total', 'executed', 'upToDate'), map(int, counts.groups())))
    processes[prefix] = dict(process, stdout=identity(OUT / (stream + '.stdout')),
                             stderr=identity(OUT / (stream + '.stderr')))
assert '> Task :android:backend-sherpa-asr:compileReleaseKotlin\n' in (OUT / 'sdk-publish.stdout').read_text()
assert '> Task :compileSherpaAsrReleaseKotlin\n' in (OUT / 'consumer-build.stdout').read_text()
assert '> Task :compileSherpaAsrDebugKotlin\n' in (OUT / 'consumer-debug-build.stdout').read_text()

published = []
source_bindings = []
metadata_bindings = []
for name in MODULES:
    directory = M2 / 'io/ondeviceai' / name / '0.1.0-SNAPSHOT'
    stem = name + '-0.1.0-SNAPSHOT'
    aar = directory / (stem + '.aar')
    source_jar = directory / (stem + '-sources.jar')
    pom = directory / (stem + '.pom')
    module = directory / (stem + '.module')
    output_aar = ROOT / 'android' / name / 'build/outputs/aar' / (name + '-release.aar')
    assert aar.read_bytes() == output_aar.read_bytes()
    with zipfile.ZipFile(aar) as archive:
        native = [p for p in archive.namelist() if p.endswith('.so')]
        assert not native
    published.append({'coordinate': 'io.ondeviceai:' + name + ':0.1.0-SNAPSHOT',
                      'aar': identity(aar), 'sourceJar': identity(source_jar), 'pom': identity(pom),
                      'moduleMetadata': identity(module), 'localOutputByteEqual': True,
                      'nativeMembers': native})
    source_root = ROOT / 'android' / name / 'src/main/kotlin'
    expected_sources = {str(p.relative_to(source_root)): p for p in source_root.rglob('*.kt')}
    with zipfile.ZipFile(source_jar) as archive:
        actual_sources = {p for p in archive.namelist() if p.endswith('.kt')}
        assert actual_sources == set(expected_sources), name
        for member, path in sorted(expected_sources.items()):
            data = archive.read(member)
            assert data == path.read_bytes()
            source_bindings.append({'module': name, 'jarMember': member,
                                    'sourcePath': str(path.relative_to(ROOT)),
                                    'bytes': len(data), 'sha256': sha(data), 'byteEqual': True})
    metadata = json.loads(module.read_text())
    assert metadata['component']['group'] == 'io.ondeviceai'
    assert metadata['component']['module'] == name
    for variant in metadata['variants']:
        for item in variant.get('files', []):
            assert item['name'] == Path(item['name']).name
            path = directory / item['name']
            data = path.read_bytes()
            assert len(data) == item['size']
            algorithms = [algorithm for algorithm in ('sha512', 'sha256', 'sha1', 'md5') if algorithm in item]
            for algorithm in algorithms:
                assert sha(data, algorithm) == item[algorithm]
            metadata_bindings.append({'module': name, 'variant': variant['name'],
                                      'file': item['name'], 'hashAlgorithmsVerified': algorithms,
                                      'sha256': sha(data)})

        resolved[configuration].append({'component': entry['component'], 'artifact': identity(path)})
