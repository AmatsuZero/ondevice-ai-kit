cmake_minimum_required(VERSION 3.22)
project(openvoice_demo LANGUAGES CXX)
add_library(onnxruntime SHARED IMPORTED)
if(DEFINED SHERPA_ONNX_JNI_DIR)
    set(onnxruntime_jni_dir "${SHERPA_ONNX_JNI_DIR}")
else()
    set(onnxruntime_jni_dir "${CMAKE_CURRENT_SOURCE_DIR}/build/android/jni")
endif()
set(onnxruntime_library "${onnxruntime_jni_dir}/${ANDROID_ABI}/libonnxruntime.so")
if(NOT EXISTS "${onnxruntime_library}")
    message(FATAL_ERROR "Missing ONNX Runtime for selected ABI ${ANDROID_ABI}: ${onnxruntime_library}")
endif()
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${onnxruntime_library}")
add_library(openvoice_demo SHARED OpenVoice.cpp AndroidOpenVoice.cpp)
target_compile_features(openvoice_demo PRIVATE cxx_std_17)
target_compile_options(openvoice_demo PRIVATE -O2)
target_link_options(openvoice_demo PRIVATE "-Wl,-z,max-page-size=16384")
target_include_directories(openvoice_demo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/build/android/headers")
target_link_libraries(openvoice_demo PRIVATE onnxruntime)
